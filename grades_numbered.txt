    17	        self.name = name

    20	    def average(self):
    21	        return sum(self.marks) / len(self.marks) if self.marks else 0

    24	    def __add__(self, other):

    27	    # Feature 8: s1 < s2 -> compare by average (lets sorted() rank students)
    28	    def __lt__(self, other):
    29	        return self.average() < other.average()

    47	def create_student(name, marks, bonus=0):
    48	    """Build a Student. Every mark gets `bonus` extra points (capped at 100).
    49	    The caller's `marks` list is NOT modified (a new list is built)."""
    50	    boosted = [min(m + bonus, 100) for m in marks]    # local variable
    51	    return Student(name, boosted)

    85	# works out a student's total, average
    86	def compute_grade(student, pass_mark=50):
    87	    """Return (total, average, letter) for one student."""
    88	    total = sum(student.marks)                                  # local
    89	    average = total / len(student.marks) if student.marks else 0  # local
    90	
    91	    def to_letter(score):                       # nested helper
    92	        if score < pass_mark:                   # reads parent's variable
    93	            return "F"
    94	        if score >= 80:
    95	            return "A"
    96	        if score >= 70:
    97	            return "B"
    98	        if score >= 60:
    99	            return "C"
   100	        return "D"
   101	
   102	    grade = to_letter(average)                  # local
   103	    return total, average, grade
   104	    # <- on return: total, average, grade, to_letter and the frame are destroyed

   140	# in / out / inout modes; pass-by-value, pass-by-result,
   141	# pass-by-value-result, pass-by-reference.

   143	# 1. Mode: IN | Method: Pass-by-value
   144	# Status in Python: NATIVE-EQUIVALENT for immutable types (int, float, str)
   145	def preview_adjusted_mark(mark, bonus):
   146	    """Caller -> Subprogram only. Rebinding `mark` locally does NOT
   147	    change the caller's variable."""
   148	    mark = min(mark + bonus, 100)       # modifies local binding only
   149	    return mark
   150	
   151	
   152	# 2. Mode: OUT | Method: Pass-by-result
   153	# Status in Python: SIMULATED (using a mutable dict populated only at exit)
   154	def compute_mark_summary(marks, out_summary):
   155	    """Subprogram -> Caller only. Ignores initial contents of `out_summary`
   156	    and writes results back only at the end of the subprogram."""
   157	    if not marks:
   158	        result = {"min": 0, "max": 0, "avg": 0.0}
   159	    else:
   160	        result = {
   161	            "min": min(marks),
   162	            "max": max(marks),
   163	            "avg": round(sum(marks) / len(marks), 1),
   164	        }
   165	    # Copy-out at the end of execution
   166	    out_summary.clear()
   167	    out_summary.update(result)
   168	
   169	
   170	# 3. Mode: INOUT | Method: Pass-by-value-result (copy-in, copy-out)
   171	# Status in Python: SIMULATED (using a 1-element list as a mutable box)
   172	def adjust_mark_value_result(mark_box, bonus):
   173	    """Caller <-> Subprogram. Copies value IN at start, works on a local
   174	    copy, and copies value OUT to caller's box only at the end."""
   175	    local_mark = mark_box[0]            # Copy-in at start
   176	    local_mark = max(0, min(local_mark + bonus, 100))
   177	    # Notice: mark_box[0] is still unchanged until the line below
   178	    mark_box[0] = local_mark            # Copy-out at end
   179	
   180	
   181	# 4. Mode: INOUT | Method: Pass-by-reference
   182	# Status in Python: NATIVE-EQUIVALENT for mutable objects (list, Student)
   183	def apply_bonus_by_ref(student, index, bonus):
   184	    """Caller <-> Subprogram. Mutates the caller's Student object in-place
   185	    through its object reference; changes are visible immediately."""
   186	    student.marks[index] = max(0, min(student.marks[index] + bonus, 100))

   296	# Feature 4b: Generic subprogram (NATIVE via dynamic typing / duck typing,
   297	#             with static TypeVar annotations from Python's typing module)
   298	def find_highest(
   299	    items: List[T],
   300	    key_fn: Callable[[T], float]
   301	) -> Optional[T]:
   302	    """Generic function that returns the highest item of any type T,
   303	    using key_fn to extract a comparable numeric score."""
   304	    if not items:
   305	        return None
   306	    best = items[0]
   307	    best_score = key_fn(best)
   308	    for item in items[1:]:
   309	        score = key_fn(item)
   310	        if score > best_score:
   311	            best = item
   312	            best_score = score
   313	    return best

   316	# Feature 5: Subprograms as parameters and indirect calls (NATIVE in Python)
   317	def compute_class_stat(
   318	    student_list: List[Student],
   319	    stat_fn: Callable[[List[float]], float]
   320	) -> float:
   321	    """Accepts a statistic function (stat_fn) as a parameter and invokes it
   322	    indirectly on the list of student averages."""
   323	    if not student_list:
   324	        return 0.0
   325	    averages = [s.average() for s in student_list]
   326	    return stat_fn(averages)            # Indirect call to passed function

   329	def apply_to_all(
   330	    student_list: List[Student],
   331	    mark_fn: Callable[[int], int]
   332	) -> None:
   333	    """Higher-order function (Feature 5) used by M5's option_curve_marks().
   334	    Accepts a transformation function and calls it indirectly on every mark."""
   335	    for s in student_list:
   336	        s.marks = [mark_fn(m) for m in s.marks]     # Indirect call
   337

   339	def _mean(values: List[float]) -> float:
   340	    """Helper statistic function passed to compute_class_stat()."""
   341	    return sum(values) / len(values)

   394	# Feature 6: Closure (NATIVE in Python)
   395	def make_curve(points, cap=100):
   396	    """Outer function ('curve maker') that returns a closure."""
   397	    def apply_curve(mark):
   398	        # Retains `points` and `cap` even after make_curve() returns
   399	        return min(mark + points, cap)
   400	    return apply_curve

   403	# Task 2: Recursive subprogram (NATIVE in Python)
   404	def total_marks_recursive(marks, index=0):
   405	    """Recursively calculates a student's total marks.
   406	    Used in the report and for Task 2 Stack Diagram #2."""
   407	    if index == len(marks):             # Base case
   408	        return 0
   409	    current = marks[index]              # Local variable in this activation record
   410	    rest_total = total_marks_recursive(marks, index + 1)
   411	    return current + rest_total

   414	# Feature 7: Generator / Semi-Coroutine (NATIVE in Python)
   415	def report_generator(student_list):
   416	    """Yields one student's report at a time, suspending execution
   417	    and preserving local state between calls."""
   418	    for idx, s in enumerate(student_list, 1):
   419	        total = total_marks_recursive(s.marks)
   420	        avg = s.average()
   421	        status = "PASS" if avg >= 50 else "FAIL"
   422	        yield (
   423	            f"[{idx}/{len(student_list)}] {s.name:<8} | "
   424	            f"Marks: {s.marks} | Total: {total} | Avg: {avg:5.1f} | {status}"
   425	        )
